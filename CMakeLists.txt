cmake_minimum_required(VERSION 3.16)
project(ExternalMemorySorting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_library(external_sort src/run_file.cpp src/buffer.cpp src/run_generation.cpp src/merge.cpp)
target_sources(external_sort PRIVATE src/loser_tree.cpp src/replacement_selection.cpp)
target_sources(external_sort PRIVATE src/optimal_merge.cpp)
find_package(Threads REQUIRED)
target_sources(external_sort PRIVATE src/pipeline.cpp)
target_link_libraries(external_sort PUBLIC Threads::Threads)
target_include_directories(external_sort PUBLIC include)
add_executable(week3_tests tests/week3_tests.cpp)
target_link_libraries(week3_tests PRIVATE external_sort)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-output/week3")
add_test(NAME week3 COMMAND week3_tests)
set_tests_properties(week3 PROPERTIES
  WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-output/week3" TIMEOUT 30)
add_executable(week3_demo apps/week3_demo.cpp)
target_link_libraries(week3_demo PRIVATE external_sort)
add_executable(week4_tests tests/week4_tests.cpp)
target_link_libraries(week4_tests PRIVATE external_sort)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-output/week4")
add_test(NAME week4 COMMAND week4_tests)
set_tests_properties(week4 PROPERTIES
  WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-output/week4" TIMEOUT 120)
add_executable(week4_demo apps/week4_demo.cpp)
target_link_libraries(week4_demo PRIVATE external_sort)
add_executable(week5_tests tests/week5_tests.cpp)
target_link_libraries(week5_tests PRIVATE external_sort)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-output/week5")
add_test(NAME week5 COMMAND week5_tests)
set_tests_properties(week5 PROPERTIES
  WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-output/week5" TIMEOUT 180)
add_executable(week5_demo apps/week5_demo.cpp)
target_link_libraries(week5_demo PRIVATE external_sort)
